<script>

var factors = [

  {id:"chf", label:"Insuficiencia cardiaca / disfunción VI", points:1},

  {id:"htn", label:"Hipertensión", points:1},

  {id:"age75", label:"Edad ≥ 75 años", points:2},

  {id:"dm", label:"Diabetes mellitus", points:1},

  {id:"stroke", label:"ACV / AIT / tromboembolismo previo", points:2},

  {id:"vasc", label:"Enfermedad vascular", points:1},

  {id:"age65", label:"Edad 65-74 años", points:1},

  {id:"female", label:"Sexo femenino", points:1}

];

var container = document.getElementById('items');

factors.forEach(function(f){

  var row = document.createElement('label');

  row.className = 'factor-row';

  row.innerHTML = '<span>'+f.label+'</span><span class="pts">+'+f.points+'</span>';

  var checkbox = document.createElement('input');

  checkbox.type = 'checkbox';

  checkbox.dataset.points = f.points;

  row.prepend(checkbox);

  container.appendChild(row);

});

function updateScore(){

  var checks = container.querySelectorAll('input[type=checkbox]');

  var total = 0;

  checks.forEach(function(c){ if(c.checked) total += parseInt(c.dataset.points); });

  document.getElementById('score').textContent = total;

  var label = document.getElementById('risk-label');

  if(total === 0){ label.textContent = "Riesgo bajo"; }

  else if(total === 1){ label.textContent = "Riesgo bajo-moderado"; }

  else { label.textContent = "Considerar anticoagulación"; }

}

container.addEventListener('change', updateScore);

</script>
